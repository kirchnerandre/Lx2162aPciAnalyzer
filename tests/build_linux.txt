Build PCIEMU inside Ubuntu 24
=============================

sudo apt update
sudo apt install build-essential
sudo apt install libglib2.0-dev libgcrypt20-dev zlib1g-dev autoconf automake libtool bison flex libpixman-1-dev
sudo apt install python3-pip
sudo apt install libvdeplug-dev
sudo apt install libncurses5-dev libncursesw5-dev
sudo apt install libssh-dev pkg-config
sudo apt-get install python3-sphinx
sudo apt install python3-sphinx-rtd-theme
sudo apt-get install ninja-build
sudo apt install slirp4netns libslirp-dev pkg-config
sudo apt install libcap-ng-dev libattr1-dev
sudo apt install libsdl2-dev
sudo apt install libgtk-3-dev

git clone https://github.com/kirchnerandre/Lx2162aPciAnalyzer.git

cd Lx2162aPciAnalyzer

git submodule update --init ./tests/qemu

cd tests

./setup.sh

cd qemu/build

make -j$(nproc)

sudo ./qemu-system-x86_64 \
    -m 16G \
    -cpu host \
    -smp 4 \
    -drive file=/mnt/d/Users/ankirchner/_Projects/vhd_server_serverdatacenter_en-us_vl_linux/26101.1068.amd64fre.ge_release_azcc_r1.250522-1447_server_serverdatacenter_en-us_vl.vhd,format=vpc \
    -machine q35,accel=kvm \
    -device VGA \
    -device lx2162a_device,id=lx2162a_device,bus=pcie.0,addr=0x10
