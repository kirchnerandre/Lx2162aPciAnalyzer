git clone https://github.com/kirchnerandre/Lx2162aPciAnalyzer.git

cd Lx2162aPciAnalyzer

git submodule update --init ./tests/qemu

cd tests

./setup.sh

cd qemu/build

make -j$(nproc)

sudo ./qemu-system-x86_64 \
    -m 16G \
    -cpu host \
    -smp 4 \
    -drive file=/mnt/c/Users/ankirchner/_Projects/ankirchner-vm-0.vhd,format=vpc \
    -machine q35,accel=kvm \
    -device VGA \
    -device lx2162a_pci_device,id=lx2162a_pci_device,bus=pcie.0,addr=0x10
